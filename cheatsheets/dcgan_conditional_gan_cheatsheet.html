# Пример кода для Основы GAN (vanilla GAN на PyTorch)
import torch
import torch.nn as nn
import torch.optim as optim

# Простые сети: полносвязный генератор и дискриминатор
latent_dim = 100
data_dim = 784  # пример: изображения 28x28, развернутые в вектор


class Generator(nn.Module):
    def __init__(self, latent_dim, data_dim):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(latent_dim, 256),
            nn.ReLU(True),
            nn.Linear(256, 512),
            nn.ReLU(True),
            nn.Linear(512, data_dim),
            nn.Tanh(),  # данные в диапазоне [-1, 1]
        )

    def forward(self, z):
        return self.net(z)


class Discriminator(nn.Module):
    def __init__(self, data_dim):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(data_dim, 512),
            nn.LeakyReLU(0.2, inplace=True),
            nn.Linear(512, 256),
            nn.LeakyReLU(0.2, inplace=True),
            nn.Linear(256, 1),
        )  # без Sigmoid: используем BCEWithLogitsLoss

    def forward(self, x):
        return self.net(x).view(-1)


device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
G = Generator(latent_dim, data_dim).to(device)
D = Discriminator(data_dim).to(device)

criterion = nn.BCEWithLogitsLoss()
optimizer_G = optim.Adam(G.parameters(), lr=2e-4, betas=(0.5, 0.999))
optimizer_D = optim.Adam(D.parameters(), lr=2e-4, betas=(0.5, 0.999))

# dataloader должен отдавать real_images размерности [batch_size, data_dim]
num_epochs = 5
for epoch in range(num_epochs):
    for real_images, _ in dataloader:
        real_images = real_images.view(real_images.size(0), -1).to(device)
        batch_size = real_images.size(0)

        # ----- Обучение дискриминатора -----
        z = torch.randn(batch_size, latent_dim, device=device)
        fake_images = G(z).detach()  # останавливаем градиент генератора

        real_labels = torch.ones(batch_size, device=device)
        fake_labels = torch.zeros(batch_size, device=device)

        D_real = D(real_images)
        D_fake = D(fake_images)

        loss_D_real = criterion(D_real, real_labels)
        loss_D_fake = criterion(D_fake, fake_labels)
        loss_D = (loss_D_real + loss_D_fake) / 2

        optimizer_D.zero_grad()
        loss_D.backward()
        optimizer_D.step()

        # ----- Обучение генератора -----
        z = torch.randn(batch_size, latent_dim, device=device)
        fake_images = G(z)
        # генератор хочет, чтобы D считал поддельные данные реальными (метки 1)
        G_labels = torch.ones(batch_size, device=device)
        D_fake_for_G = D(fake_images)
        loss_G = criterion(D_fake_for_G, G_labels)

        optimizer_G.zero_grad()
        loss_G.backward()
        optimizer_G.step()

    print(f"Epoch {epoch+1}: loss_D={loss_D.item():.3f}, loss_G={loss_G.item():.3f}")

# Правила архитектуры DCGAN (Radford et al., 2015)
# • вместо пулинга — свёртки со stride 2 (D) и транспонированные свёртки (G)
# • BatchNorm в G и D (кроме выхода G и входа D)
# • без полносвязных скрытых слоёв
# • G: ReLU, на выходе Tanh;  D: LeakyReLU(0.2)
import torch.nn as nn

def weights_init(m):
    """Инициализация весов из статьи DCGAN: N(0, 0.02)."""
    if isinstance(m, (nn.Conv2d, nn.ConvTranspose2d)):
        nn.init.normal_(m.weight, 0.0, 0.02)
    elif isinstance(m, nn.BatchNorm2d):
        nn.init.normal_(m.weight, 1.0, 0.02)
        nn.init.zeros_(m.bias)

import torch
import torch.nn as nn

class Generator(nn.Module):
    """z (nz×1×1) → изображение 3×64×64 в диапазоне [-1, 1]."""
    def __init__(self, nz=100, ngf=64, nc=3):
        super().__init__()
        self.net = nn.Sequential(
            nn.ConvTranspose2d(nz, ngf * 8, 4, 1, 0, bias=False),      # 4×4
            nn.BatchNorm2d(ngf * 8), nn.ReLU(True),
            nn.ConvTranspose2d(ngf * 8, ngf * 4, 4, 2, 1, bias=False), # 8×8
            nn.BatchNorm2d(ngf * 4), nn.ReLU(True),
            nn.ConvTranspose2d(ngf * 4, ngf * 2, 4, 2, 1, bias=False), # 16×16
            nn.BatchNorm2d(ngf * 2), nn.ReLU(True),
            nn.ConvTranspose2d(ngf * 2, ngf, 4, 2, 1, bias=False),     # 32×32
            nn.BatchNorm2d(ngf), nn.ReLU(True),
            nn.ConvTranspose2d(ngf, nc, 4, 2, 1, bias=False),          # 64×64
            nn.Tanh(),
        )

    def forward(self, z):
        return self.net(z)

z = torch.randn(16, 100, 1, 1)
fake = Generator()(z)          # torch.Size([16, 3, 64, 64])

import torch.nn as nn

class Discriminator(nn.Module):
    """Изображение 3×64×64 → вероятность «настоящее»."""
    def __init__(self, ndf=64, nc=3):
        super().__init__()
        self.net = nn.Sequential(
            nn.Conv2d(nc, ndf, 4, 2, 1, bias=False),                   # 32×32
            nn.LeakyReLU(0.2, inplace=True),
            nn.Conv2d(ndf, ndf * 2, 4, 2, 1, bias=False),              # 16×16
            nn.BatchNorm2d(ndf * 2), nn.LeakyReLU(0.2, inplace=True),
            nn.Conv2d(ndf * 2, ndf * 4, 4, 2, 1, bias=False),          # 8×8
            nn.BatchNorm2d(ndf * 4), nn.LeakyReLU(0.2, inplace=True),
            nn.Conv2d(ndf * 4, ndf * 8, 4, 2, 1, bias=False),          # 4×4
            nn.BatchNorm2d(ndf * 8), nn.LeakyReLU(0.2, inplace=True),
            nn.Conv2d(ndf * 8, 1, 4, 1, 0, bias=False),                # 1×1
            nn.Sigmoid(),   # с nn.BCEWithLogitsLoss сигмоиду убирают
        )

    def forward(self, x):
        return self.net(x).view(-1)

import torch
import torch.nn as nn

class CondGenerator(nn.Module):
    """G(z, y): к шуму добавляется эмбеддинг метки класса."""
    def __init__(self, nz=100, n_classes=10, img_dim=28 * 28):
        super().__init__()
        self.label_emb = nn.Embedding(n_classes, n_classes)
        self.net = nn.Sequential(
            nn.Linear(nz + n_classes, 256), nn.LeakyReLU(0.2),
            nn.Linear(256, 512), nn.BatchNorm1d(512), nn.LeakyReLU(0.2),
            nn.Linear(512, img_dim), nn.Tanh(),
        )

    def forward(self, z, labels):
        return self.net(torch.cat([z, self.label_emb(labels)], dim=1))

class CondDiscriminator(nn.Module):
    """D(x, y): получает и изображение, и метку."""
    def __init__(self, n_classes=10, img_dim=28 * 28):
        super().__init__()
        self.label_emb = nn.Embedding(n_classes, n_classes)
        self.net = nn.Sequential(
            nn.Linear(img_dim + n_classes, 512), nn.LeakyReLU(0.2),
            nn.Linear(512, 256), nn.LeakyReLU(0.2),
            nn.Linear(256, 1), nn.Sigmoid(),
        )

    def forward(self, img, labels):
        return self.net(torch.cat([img.view(img.size(0), -1), self.label_emb(labels)], dim=1))

import torch
import torch.nn.functional as F

# Стандартная GAN (BCE на логитах)
def d_loss_bce(real_logits, fake_logits):
    return (F.binary_cross_entropy_with_logits(real_logits, torch.ones_like(real_logits)) +
            F.binary_cross_entropy_with_logits(fake_logits, torch.zeros_like(fake_logits)))

def g_loss_bce(fake_logits):                      # non-saturating: max log D(G(z))
    return F.binary_cross_entropy_with_logits(fake_logits, torch.ones_like(fake_logits))

# Wasserstein GAN с gradient penalty (WGAN-GP): у критика нет сигмоиды
def d_loss_wgan_gp(critic, real, fake, lam=10.0):
    eps = torch.rand(real.size(0), 1, 1, 1, device=real.device)
    mix = (eps * real + (1 - eps) * fake).requires_grad_(True)
    grad = torch.autograd.grad(critic(mix).sum(), mix, create_graph=True)[0]
    gp = ((grad.flatten(1).norm(2, dim=1) - 1) ** 2).mean()
    return critic(fake).mean() - critic(real).mean() + lam * gp

def g_loss_wgan(critic, fake):
    return -critic(fake).mean()

import torch
import torch.nn as nn

# Диагностика: насколько различаются сгенерированные образцы
def diversity(fake):
    flat = fake.flatten(1)
    return torch.cdist(flat, flat).mean().item()   # резкое падение — признак mode collapse

# Решение: minibatch discrimination (Salimans et al., 2016) —
# D видит статистику по батчу и замечает одинаковые образцы
class MinibatchDiscrimination(nn.Module):
    def __init__(self, in_features, out_features=50, kernel_dims=5):
        super().__init__()
        self.T = nn.Parameter(torch.randn(in_features, out_features, kernel_dims) * 0.1)

    def forward(self, x):                                    # x: (B, in_features)
        M = torch.einsum("bi,iok->bok", x, self.T)           # (B, out, k)
        l1 = (M.unsqueeze(0) - M.unsqueeze(1)).abs().sum(3)  # (B, B, out)
        o = torch.exp(-l1).sum(1) - 1                        # без сравнения с собой
        return torch.cat([x, o], dim=1)

import numpy as np
from scipy import linalg

def fid(mu1, sigma1, mu2, sigma2):
    """Fréchet Inception Distance между двумя наборами признаков Inception-v3.
    mu, sigma — среднее и ковариация 2048-мерных признаков. Меньше — лучше."""
    covmean = linalg.sqrtm(sigma1 @ sigma2)
    covmean = covmean.real if np.iscomplexobj(covmean) else covmean   # убираем численный мусор
    return float(((mu1 - mu2) ** 2).sum() + np.trace(sigma1 + sigma2 - 2 * covmean))

def inception_score(p_yx, eps=1e-12):
    """p_yx: (N, классы) — предсказания Inception для сгенерированных изображений. Больше — лучше."""
    p_y = p_yx.mean(axis=0, keepdims=True)
    kl = (p_yx * (np.log(p_yx + eps) - np.log(p_y + eps))).sum(axis=1)
    return float(np.exp(kl.mean()))
